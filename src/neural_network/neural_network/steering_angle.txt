import numpy as np 
import os
import csv
import cv2
import torch
import torch.nn as nn
from torch.utils.data import Dataset, DataLoader


#DataProcessing

class DrivingDataset(Dataset):
    def __init__(self, dataset_dir):
        self.dataset_dir = dataset_dir
        self.samples = []

        csv_path = os.path.join(dataset_dir, 'data.csv')
        with open(csv_path, 'r') as file:
            reader = csv.DictReader(file)
            for row in reader:
                self.samples.append((row['image_path'], float(row['steering_angle'])))

    def __len__(self):
        return len(self.samples)

    def __getitem__(self, idx):
         rel_path, steering = self.samples[idx]
         full_path = os.path.join(self.dataset_dir, rel_path)

         img = cv2.imread(full_path)
         if img is None:
             raise FileNotFoundError(f"Could not load image at {full_path}")

         cropped = img[180:440, :]
         resized = cv2.resize(cropped, (200, 66), interpolation=cv2.INTER_AREA)

         img_normalized = (resized/127.5) - 1.0
         img_chw = np.transpose(img_normalized, (2, 0, 1))

         return torch.tensor(img_chw, dtype=torch.float32), torch.tensor([steering], dtype=torch.float32)

class CarNN(nn.Module):
    def __init__(self):
        super(CarNN, self).__init__()
        
        # 1. Convolutional Layers (Feature Extraction)
        self.conv_layers = nn.Sequential(
            nn.Conv2d(3, 24, kernel_size=5, stride=2),
            nn.ELU(),
            nn.Conv2d(24, 36, kernel_size=5, stride=2),
            nn.ELU(),
            nn.Conv2d(36, 48, kernel_size=5, stride=2),
            nn.ELU(),
            nn.Conv2d(48, 64, kernel_size=3, stride=1),
            nn.ELU(),
            nn.Conv2d(64, 64, kernel_size=3, stride=1),
            nn.ELU())
        
        # 2. Fully Connected Layers (Decision Making)
        self.fc_layers = nn.Sequential(
            nn.Flatten(),
            nn.Linear(1152, 100),
            nn.ELU(),
            nn.Linear(100, 50),
            nn.ELU(),
            nn.Linear(50, 10),
            nn.ELU(),
            nn.Linear(10, 1)  # Final output: 1 single steering angle
        )

    def forward(self, x):
        features = self.conv_layers(x)
        steering_prediction = self.fc_layers(features)
        return steering_prediction
    
def train():

    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    print(f"Training on device: {device}")

    dataset_path = os.path.expanduser('~/autonomous_car_nn/dataset')
    dataset = DrivingDataset(dataset_path)

    dataloader = DataLoader(dataset, batch_size = 32, shuffle=True)


    print(f"Total samples in dataset: {len(dataset)}")

    img, actual_angle = dataset[0]
    batched_img = img.unsqueeze(0)

    model = CarNN().to(device)
    criterion = nn.MSELoss()
    optimizer = torch.optim.Adam(model.parameters(), lr=0.001)

    print(f"Starting training on {len(dataset)} samples...")

    num_epochs = 5

    for epoch in range(num_epochs):
        running_loss = 0.0

        for i, (images, true_angles) in enumerate(dataloader):
            images = images.to(device)
            true_angles = true_angles.to(device)

            predicted_angles = model(images)
            loss = criterion(predicted_angles, true_angles)
            optimizer.zero_grad()
            loss.backward()
            optimizer.step()

            running_loss += loss.item()

            if(i + 1) % 50 == 0:
                print(f"Epoch [{epoch+1}/{num_epochs}], BATCH[{i+1}/{len(dataloader)}], Loss: {loss.item():.4f}")

        avg_loss = running_loss / len(dataloader)
        print(f"--- End of Epoch {epoch+1}, Average Loss: {avg_loss:.4f}---")

    model_save_path = os.path.expanduser('~/autonomous_car_nn/autopilot_brain.pth')
    torch.save(model.state_dict(), model_save_path)
    print(f"Training Completed! Model saved to {model_save_path}")



    predicted_angle = model(batched_img)

    print(f"Image Tensor Shape: {img.shape}")
    print(f"Network Output Shape: {predicted_angle.shape}")
    print(f"Predicted Angle (Random Initialization): {predicted_angle.item():.4f}")
    print(f"Steering Tensor: {actual_angle.item():.4f}")

if __name__ == '__main__':
    train()
